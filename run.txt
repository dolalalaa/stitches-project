add products:
curl -X POST http://localhost:1206/product/create \
  -H "Content-Type: application/json" \
  -d "{\"name\":\"Lavender Chikankari Kurti\",\"type\":\"Customized\",\"stock\":7,\"price\":7149,\"image\":\"YOUR_POSTIMAGES_LINK_HERE\"}"


delete product:
curl -X DELETE http://localhost:1206/product/FIRST_ID_HERE


run url to generate mannequin:
http://localhost:5173/mannequin?shoulder=80&chest=90&waist=70&hip=95&armLength=50
